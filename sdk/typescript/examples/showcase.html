// 1. Enriched League Standings + Form Table (with goal splits, rank changes & promotions)

// 2. Club Intelligence Bundle (profile, 41-man squad, active missing/injured players, 3-5-2 lineup & season stats)
const milan = await client.getTeamIntelligenceBundle(2692, { seasonId: 130971 });

// 3. Tennis Player Intelligence Bundle (profile, ATP rank & 2004–2026 surface W-L matrix)
const djokovic = await client.getTennisPlayerIntelligenceBundle(14882);

// 4. First-to-Score Goal Timing, Half Over/Under Splits & NBA Team Pace

  client.getSeasonGoalsDistribution(140782),
  client.getSeasonOverUnder(140782),
